    @classmethod
    def from_local(cls, ckpt_dir, device, t3_filename: str = None, s3gen_filename: str = None,
                   ve_path=None, tokenizer_path=None) -> 'ChatterboxTTS':
        """Load from explicit paths. Only .safetensors weights are read (no pickle)."""
        ckpt_dir = Path(ckpt_dir)
        t3_path = ckpt_dir / (t3_filename or T3_FILENAME)
        s3gen_path = ckpt_dir / (s3gen_filename or S3GEN_FILENAME)
        ve_path = Path(ve_path) if ve_path else ckpt_dir / VE_FILENAME
        tokenizer_path = Path(tokenizer_path) if tokenizer_path else ckpt_dir / TOKENIZER_FILENAME
        for p in (t3_path, s3gen_path, ve_path):
            if p.suffix != ".safetensors":
                raise ValueError(f"Refusing to load non-safetensors weights: {p}")

        ve = VoiceEncoder()
        ve.load_state_dict(load_safetensors(str(ve_path)))
        ve.to(device).eval()

        t3_cfg = T3ConfigMultilingual()
        t3_cfg.text_tokens_dict_size = T3_TEXT_VOCAB_SIZE
        t3 = T3(t3_cfg)
        t3_state = load_safetensors(str(t3_path))
        if "model" in t3_state.keys():
            t3_state = t3_state["model"][0]
        t3.load_state_dict(t3_state)
        t3.to(device).eval()

        s3gen = S3Gen()
        s3gen.load_state_dict(load_safetensors(str(s3gen_path)), strict=False)
        s3gen.to(device).eval()

        tokenizer = MTLTokenizer(str(tokenizer_path))
        # No builtin conds.pt: the pipeline always conditions on the user's reference wav.
        return cls(t3, s3gen, ve, tokenizer, device, conds=None)

    @classmethod
    def from_pretrained(cls, device: torch.device) -> 'ChatterboxTTS':
        token = os.getenv("HF_TOKEN")
        for repo in (REPO_ID, BASE_REPO_ID):
            if not repo.startswith(TRUSTED_ORG + "/"):
                raise ValueError(f"Untrusted model repo: {repo}")

        def fetch(repo_id, filename):
            return Path(hf_hub_download(repo_id=repo_id, filename=filename,
                                        repo_type="model", revision="main", token=token))

        t3_path = fetch(REPO_ID, T3_FILENAME)
        s3gen_path = fetch(REPO_ID, S3GEN_FILENAME)
        tokenizer_path = fetch(REPO_ID, TOKENIZER_FILENAME)
        ve_path = fetch(BASE_REPO_ID, VE_FILENAME)
        if t3_path.parent != s3gen_path.parent:
            raise RuntimeError("T3 and S3Gen resolved to different snapshots")
        return cls.from_local(t3_path.parent, device, T3_FILENAME, S3GEN_FILENAME,
                              ve_path=ve_path, tokenizer_path=tokenizer_path)
